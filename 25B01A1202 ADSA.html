#include <stdio.h>
int Knapsack(int weights[],int profits[],int n,int m){
	int M[n+1][m+1];
	for(int i=0;i<=n;i++){
		for(int j=0;j<=m;j++){
			if(i == 0||j == 0)
				M[i][j] = 0;
			else if(weights[i-1] <= j)
				M[i][j]=(profits[i-1]+M[i-1][j-weights[i-1]]>M[i-1][j])?(profits[i-1]+M[i-1][j-weights[i-1]]):M[i-1][j];
			else
				M[i][j] = M[i-1][j];
		}
	}
	return M[n][m];
}
